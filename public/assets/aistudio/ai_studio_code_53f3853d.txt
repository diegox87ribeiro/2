      - name: Restaurar ou gerar debug.keystore do projeto
        run: |
          if [ ! -s debug.keystore ] && [ -s debug.keystore.base64 ]; then
            base64 -d debug.keystore.base64 > debug.keystore
          fi
          if [ ! -s debug.keystore ]; then
            keytool -genkeypair -v \
              -keystore debug.keystore \
              -storepass android \
              -alias androiddebugkey \
              -keypass android \
              -keyalg RSA \
              -keysize 2048 \
              -validity 10000 \
              -dname "CN=Android Debug,O=Android,C=US"
          fi
